## 流水线异常告警通知脚本

### 1.1 定位与触发机制

- **定位**：Mac 生产节点本地告警触发器，对接流水线失败分支，实时推送异常通知
- **触发方式**：
  1. **主触发**：流水线执行失败时自动调用，毫秒级响应
  2. **兜底巡检**：定时扫描批次目录，发现漏网失败批次补告警
- **通知渠道**：企业微信机器人（优先）+ 邮件（备用扩展）
- **去重机制**：同一批次 10 分钟内仅告警一次，避免消息轰炸
- **信息维度**：批次ID、错误阶段、节点标识、发生时间、快捷排查链接、错误摘要

### 1.2 完整脚本

文件路径：`scripts/pipeline-tools/alert_notify.py`

```
#!/usr/bin/env python
# ==============================================================
# H3 数字人流水线异常告警通知脚本 v1.0
# 对接：全链路流水线 + A2A 审计体系
# 渠道：企业微信机器人
# 规范：YYC³ 运维告警分级 + 全链路可追溯
# ==============================================================
import os
import sys
import json
import time
import argparse
import urllib.request
from datetime import datetime

# -------------------------- 配置 --------------------------
ALARM_CONFIG = {
    # 企业微信机器人 Webhook 地址
    "wecom_webhook": "https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=你的机器人key",
    # 告警去重窗口（秒）
    "dedup_window": 600,
    # 生产节点标识
    "node_name": "mac-pro-h3-01",
    # 面板跳转地址
    "dashboard_url": "https://api.0379.world/h3-dashboard/",
    # 告警级别对应颜色
    "level_color": {
        "critical": "warning",
        "error": "warning",
        "warning": "comment",
        "info": "info"
    }
}

# 告警缓存目录（去重用）
CACHE_DIR = os.path.join(os.path.dirname(os.path.abspath(__file__)), ".alert_cache")
os.makedirs(CACHE_DIR, exist_ok=True)

# -------------------------- 核心方法 --------------------------
def send_wecom(title: str, content: str, level: str = "error") -> bool:
    """发送企业微信机器人 Markdown 消息"""
    webhook = ALARM_CONFIG["wecom_webhook"]
    if not webhook or "你的机器人key" in webhook:
        print("[告警] 未配置企业微信 Webhook，跳过发送")
        return False

    msg = {
        "msgtype": "markdown",
        "markdown": {
            "content": f"""<font color="{ALARM_CONFIG['level_color'].get(level, 'warning')}">**{title}**</font>

> **节点**：{ALARM_CONFIG['node_name']}
> **时间**：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}

{content}

[👉 前往面板查看详情]({ALARM_CONFIG['dashboard_url']})
"""
        }
    }

    try:
        data = json.dumps(msg).encode("utf-8")
        req = urllib.request.Request(webhook, data=data, headers={"Content-Type": "application/json"})
        with urllib.request.urlopen(req, timeout=10) as resp:
            result = json.loads(resp.read().decode())
            return result.get("errcode") == 0
    except Exception as e:
        print(f"[告警] 发送失败：{str(e)}")
        return False

def check_duplicate(batch_id: str) -> bool:
    """去重检查：返回 True 表示窗口期内已告警过"""
    cache_file = os.path.join(CACHE_DIR, f"{batch_id}.alarm")
    if os.path.exists(cache_file):
        mtime = os.path.getmtime(cache_file)
        if time.time() - mtime < ALARM_CONFIG["dedup_window"]:
            return True
    # 写入缓存标记
    with open(cache_file, "w") as f:
        f.write(datetime.now().isoformat())
    return False

def alert_batch_failed(batch_id: str, error_msg: str,
                      manifest_path: str = None, level: str = "error"):
    """批次生产失败告警入口"""
    if check_duplicate(batch_id):
        print(f"[告警] 批次 {batch_id} 已在窗口期内告警，跳过")
        return

    # 读取 manifest 补充上下文
    extra_info = ""
    if manifest_path and os.path.exists(manifest_path):
        try:
            with open(manifest_path, "r", encoding="utf-8") as f:
                m = json.load(f)
            extra_info = f"""
> **模式**：{m.get('mode', '-')}
> **帧数**：{m.get('num_frames', '-')}
> **输入文案**：{m.get('input_text', '-')[:50]}...
"""
        except:
            pass

    content = f"""
> **批次ID**：`{batch_id}`
> **错误阶段**：流水线执行失败
{extra_info}
> **错误摘要**：
```

{error_msg[:300]}

```

**排查建议**：
1. 检查模型文件与参考图完整性
2. 查看节点内存占用与磁盘空间
3. 核对帧数是否满足 %17==5 规则
"""

    title = "⚠️ H3 数字人流水线异常告警"
    send_wecom(title, content, level)
    print(f"[告警] 批次 {batch_id} 告警已发送")

def alert_system_error(error_msg: str, level: str = "critical"):
    """系统级异常告警（节点离线、NAS挂载失败、服务宕机等）"""
    content = f"""
> **错误类型**：系统级异常
> **错误详情**：{error_msg}

请立即检查节点状态、存储挂载与服务可用性。
"""
    send_wecom("🚨 H3 生产节点系统告警", content, level)

# -------------------------- 命令行入口 --------------------------
if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="H3 流水线告警通知工具")
    parser.add_argument("--type", default="batch", choices=["batch", "system"], help="告警类型")
    parser.add_argument("--batch_id", help="批次唯一ID")
    parser.add_argument("--error", required=True, help="错误信息摘要")
    parser.add_argument("--manifest", help="manifest 文件路径")
    parser.add_argument("--level", default="error", choices=["critical", "error", "warning", "info"])

    args = parser.parse_args()

    if args.type == "batch":
        if not args.batch_id:
            print("❌ 批次告警必须指定 --batch_id")
            sys.exit(1)
        alert_batch_failed(args.batch_id, args.error, args.manifest, args.level)
    else:
        alert_system_error(args.error, args.level)
```

    run_cmd(f'python {DASHBOARD_SCRIPT} --batch_id {args.batch_id} --batch_dir {batch_dir} --status failed --mode {args.mode}')

    # 触发异常告警

    sys.exit(1)
