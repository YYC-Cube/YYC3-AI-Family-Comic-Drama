# 告警处理闭环与工单对接脚本设计

**100% 复用现有告警聚合中心、Redis 基础设施与 NAS 归档体系**，补全「告警→工单→处理→验证→闭环→报表」完整治理链路，对齐「五高架构」高安全、可审计、可追溯标准。

---

## 一、告警处理闭环与工单对接脚本

### 1.1 定位与设计规范

- **定位**：在原有多节点告警聚合中心基础上扩展工单管理模块，实现**「告警自动触发工单→分级认领→处理验证→闭环归档→根因分析」**全链路闭环，零新增基础设施。
- **设计原则**：
  - **自动建单**：error 及以上级别告警自动生成工单，warning 级别汇总建单
  - **状态联动**：工单认领后自动暂停升级计时，处理完成后触发验证
  - **根因沉淀**：闭环强制填写根因分类，定期沉淀到知识库
  - **接口兼容**：内置企业微信工单卡片通知，预留飞书/钉钉工单对接接口
  - **全量归档**：工单全生命周期写入 NAS RAID1，永久可追溯

### 1.2 工单模型规范

| 字段 | 类型 | 说明 |
| --- | --- | --- |
| `ticket_id` | string | 工单唯一ID，`TK-时间戳-随机串` |
| `alert_id` | string | 关联告警ID |
| `level` | string | 告警级别：warning/error/critical |
| `title` | string | 工单标题 |
| `description` | string | 告警详情与现象 |
| `node` | string | 发生节点 |
| `batch_id` | string | 关联批次（可选） |
| `status` | string | 状态：`pending`待认领 / `processing`处理中 / `verify`待验证 / `closed`已闭环 / `rejected`已驳回 |
| `owner` | string | 认领人 |
| `root_cause` | string | 根因分类：配置/资源/模型/网络/存储/其他 |
| `solution` | string | 处理方案与结果 |
| `create_time` | int | 创建时间戳 |
| `handle_time` | int | 认领时间戳 |
| `close_time` | int | 闭环时间戳 |
| `sla_cost` | int | 处理耗时（秒） |

### 1.3 完整工单管理脚本

**部署位置**：DGX 节点 2，`/data/ops/ticket_manager.py`，与告警聚合中心同机部署，复用 Redis 连接。

```
#!/usr/bin/env python3
# ==============================================================
# H3 生产告警工单管理系统 v1.0
# 对接：告警聚合中心 + 企业微信通知 + NAS 归档
# 闭环：告警触发→自动建单→认领处理→验证闭环→根因沉淀
# ==============================================================
import os
import json
import time
import uuid
from datetime import datetime
import redis

# -------------------------- 配置 --------------------------
CONFIG = {
    "redis_host": "10.0.0.12",
    "redis_port": 6379,
    # 工单存储 key 前缀
    "ticket_key_prefix": "ops:ticket:",
    # 工单列表索引
    "ticket_index_key": "ops:ticket:index",
    # 企业微信 webhook
    "wecom_webhook": "https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=你的机器人key",
    # NAS 归档目录
    "nas_archive_dir": "/mnt/nas/raid1-core/h3-security/ticket-archive/",
    # SLA 时效（秒），对应告警级别
    "sla": {
        "warning": 7200,    # 2小时
        "error": 1800,      # 30分钟
        "critical": 900      # 15分钟
    },
    # 根因分类字典
    "root_cause_list": [
        "配置错误", "资源不足", "模型异常", "网络故障",
        "存储挂载异常", "NAS同步延迟", "参数非法", "依赖缺失", "其他"
    ]
}

redis_client = redis.Redis(
    host=CONFIG["redis_host"],
    port=CONFIG["redis_port"],
    decode_responses=True
)

# -------------------------- 工具函数 --------------------------
def send_wecom_card(ticket: dict, action: str = "创建"):
    """发送企业微信工单卡片通知"""
    color_map = {
        "warning": "comment",
        "error": "warning",
        "critical": "warning"
    }
    status_text = {
        "pending": "待认领",
        "processing": "处理中",
        "verify": "待验证",
        "closed": "已闭环",
        "rejected": "已驳回"
    }

    content = f"""<font color="{color_map.get(ticket['level'], 'warning')}">**【工单{action}】{ticket['title']}**</font>

> **工单ID**：`{ticket['ticket_id']}`
> **告警级别**：{ticket['level'].upper()}
> **当前状态**：{status_text.get(ticket['status'], ticket['status'])}
> **发生节点**：{ticket['node']}
> **创建时间**：{datetime.fromtimestamp(ticket['create_time']).strftime('%Y-%m-%d %H:%M:%S')}
> **SLA 时效**：{int(CONFIG['sla'].get(ticket['level'], 3600)/60)} 分钟

**问题描述：**
{ticket['description'][:200]}...

[👉 立即处理](https://api.0379.world/h3-dashboard/)
"""

    try:
        import urllib.request
        msg = {"msgtype": "markdown", "markdown": {"content": content}}
        data = json.dumps(msg).encode("utf-8")
        req = urllib.request.Request(CONFIG["wecom_webhook"], data=data, headers={"Content-Type": "application/json"})
        with urllib.request.urlopen(req, timeout=10) as resp:
            return json.loads(resp.read()).get("errcode") == 0
    except Exception as e:
        print(f"[工单通知] 发送失败：{e}")
        return False

# -------------------------- 工单管理器 --------------------------
class TicketManager:
    def __init__(self):
        os.makedirs(CONFIG["nas_archive_dir"], exist_ok=True)

    def create_ticket(self, alert_data: dict) -> dict:
        """从告警自动创建工单"""
        ticket_id = f"TK-{int(time.time())}-{uuid.uuid4().hex[:6]}"
        level = alert_data.get("level", "error")
        title = alert_data.get("title", "生产告警")
        description = alert_data.get("content", "")
        node = alert_data.get("node", "unknown")
        batch_id = alert_data.get("batch_id")

        ticket = {
            "ticket_id": ticket_id,
            "alert_id": alert_data.get("alert_id", ""),
            "level": level,
            "title": title,
            "description": description,
            "node": node,
            "batch_id": batch_id,
            "status": "pending",
            "owner": "",
            "root_cause": "",
            "solution": "",
            "create_time": int(time.time()),
            "handle_time": 0,
            "close_time": 0,
            "sla_cost": 0
        }

        # 写入 Redis
        key = f"{CONFIG['ticket_key_prefix']}{ticket_id}"
        redis_client.set(key, json.dumps(ticket, ensure_ascii=False))
        redis_client.lpush(CONFIG["ticket_index_key"], ticket_id)

        # 发送通知
        send_wecom_card(ticket, "创建")

        print(f"[工单] 已创建 {ticket_id} | {level} | {title}")
        return ticket

    def claim_ticket(self, ticket_id: str, owner: str) -> dict:
        """认领工单，暂停升级计时"""
        key = f"{CONFIG['ticket_key_prefix']}{ticket_id}"
        raw = redis_client.get(key)
        if not raw:
            return {"success": False, "error": "工单不存在"}

        ticket = json.loads(raw)
        if ticket["status"] != "pending":
            return {"success": False, "error": f"工单状态为 {ticket['status']}，无法认领"}

        ticket["status"] = "processing"
        ticket["owner"] = owner
        ticket["handle_time"] = int(time.time())

        redis_client.set(key, json.dumps(ticket, ensure_ascii=False))
        send_wecom_card(ticket, f"已认领 by {owner}")

        # 联动告警聚合中心：取消该告警升级计时
        # （通过修改告警状态实现，与聚合中心升级检查联动）
        alert_id = ticket["alert_id"]
        if alert_id:
            redis_client.hset("ops:alert:state", f"{alert_id}.status", "processing")

        print(f"[工单] {ticket_id} 已被 {owner} 认领")
        return {"success": True, "ticket": ticket}

    def resolve_ticket(self, ticket_id: str, solution: str, root_cause: str = "其他") -> dict:
        """提交处理结果，进入待验证状态"""
        key = f"{CONFIG['ticket_key_prefix']}{ticket_id}"
        raw = redis_client.get(key)
        if not raw:
            return {"success": False, "error": "工单不存在"}

        ticket = json.loads(raw)
        if ticket["status"] != "processing":
            return {"success": False, "error": f"工单状态为 {ticket['status']}，无法提交处理"}

        if root_cause not in CONFIG["root_cause_list"]:
            root_cause = "其他"

        ticket["status"] = "verify"
        ticket["solution"] = solution
        ticket["root_cause"] = root_cause

        redis_client.set(key, json.dumps(ticket, ensure_ascii=False))
        send_wecom_card(ticket, "待验证")

        print(f"[工单] {ticket_id} 提交处理，待验证")
        return {"success": True, "ticket": ticket}

    def close_ticket(self, ticket_id: str, verified: bool = True, comment: str = "") -> dict:
        """验证闭环，归档到 NAS"""
        key = f"{CONFIG['ticket_key_prefix']}{ticket_id}"
        raw = redis_client.get(key)
        if not raw:
            return {"success": False, "error": "工单不存在"}

        ticket = json.loads(raw)
        if ticket["status"] != "verify":
            return {"success": False, "error": f"工单状态为 {ticket['status']}，无法闭环"}

        if verified:
            ticket["status"] = "closed"
        else:
            ticket["status"] = "processing"
            ticket["solution"] += f"\n验证驳回：{comment}"
            redis_client.set(key, json.dumps(ticket, ensure_ascii=False))
            send_wecom_card(ticket, "验证驳回")
            return {"success": True, "ticket": ticket, "action": "rejected"}

        ticket["close_time"] = int(time.time())
        ticket["sla_cost"] = ticket["close_time"] - ticket["create_time"]

        redis_client.set(key, json.dumps(ticket, ensure_ascii=False))

        # 归档到 NAS
        self._archive_ticket(ticket)

        send_wecom_card(ticket, "已闭环")
        print(f"[工单] {ticket_id} 已闭环，耗时 {ticket['sla_cost']}s")
        return {"success": True, "ticket": ticket}

    def _archive_ticket(self, ticket: dict):
        """归档工单到 NAS RAID1"""
        month = datetime.fromtimestamp(ticket["create_time"]).strftime("%Y%m")
        archive_file = os.path.join(CONFIG["nas_archive_dir"], f"tickets-{month}.jsonl")
        with open(archive_file, "a", encoding="utf-8") as f:
            f.write(json.dumps(ticket, ensure_ascii=False) + "\n")

    def get_ticket(self, ticket_id: str) -> dict:
        """查询工单详情"""
        key = f"{CONFIG['ticket_key_prefix']}{ticket_id}"
        raw = redis_client.get(key)
        if not raw:
            return {}
        return json.loads(raw)

    def list_tickets(self, status: str = "all", limit: int = 20) -> list:
        """工单列表查询"""
        ticket_ids = redis_client.lrange(CONFIG["ticket_index_key"], 0, limit * 3)
        tickets = []
        for tid in ticket_ids:
            t = self.get_ticket(tid)
            if t and (status == "all" or t["status"] == status):
                tickets.append(t)
                if len(tickets) >= limit:
                    break
        return tickets

# -------------------------- 与告警聚合中心对接 --------------------------
# 在原有告警聚合中心 process_alert 方法中追加自动建单逻辑：
# 插入到 process_alert 方法末尾
"""
# 自动创建工单
if level in ["error", "critical"]:
    ticket_mgr = TicketManager()
    ticket_mgr.create_ticket({
        "alert_id": alert_id,
        "title": title,
        "content": content,
        "level": level,
        "node": node,
        "batch_id": batch_id
    })
"""

# -------------------------- 命令行入口 --------------------------
if __name__ == "__main__":
    import argparse
    parser = argparse.ArgumentParser(description="告警工单管理工具")
    parser.add_argument("action", choices=["create", "claim", "resolve", "close", "list", "get"])
    parser.add_argument("--ticket_id", help="工单ID")
    parser.add_argument("--owner", help="认领人")
    parser.add_argument("--solution", help="处理方案")
    parser.add_argument("--root_cause", help="根因分类")
    parser.add_argument("--status", default="all", help="状态筛选")
    parser.add_argument("--title", help="测试工单标题")
    parser.add_argument("--content", help="测试工单内容")
    parser.add_argument("--level", default="error", help="告警级别")

    args = parser.parse_args()
    mgr = TicketManager()

    if args.action == "create":
        mgr.create_ticket({"title": args.title, "content": args.content, "level": args.level, "node": "test"})
    elif args.action == "claim":
        print(mgr.claim_ticket(args.ticket_id, args.owner))
    elif args.action == "resolve":
        print(mgr.resolve_ticket(args.ticket_id, args.solution, args.root_cause))
    elif args.action == "close":
        print(mgr.close_ticket(args.ticket_id, verified=True))
    elif args.action == "list":
        for t in mgr.list_tickets(args.status):
            print(f"{t['ticket_id']} | {t['status']} | {t['level']} | {t['title']}")
    elif args.action == "get":
        print(json.dumps(mgr.get_ticket(args.ticket_id), ensure_ascii=False, indent=2))
```
