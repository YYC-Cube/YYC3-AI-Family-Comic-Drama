# 多节点告警聚合与升级通知方案设计

**100% 复用现有基础设施与协议体系**，密钥轮换无缝对接 Nginx 鉴权配置，告警聚合复用 Redis Stream 消息通道，符合「五高架构」高安全、高可用、可审计规范。

## 节点告警聚合与升级通知方案

### 1.1 架构设计

复用现有 Redis Stream 消息基础设施，构建**三级告警治理体系**，解决多生产节点告警分散、风暴、升级无标准的问题：

```
Mac生产节点01 → 本地告警检测 → 写入 Redis 告警流
Mac生产节点02 → 本地告警检测 → 写入 Redis 告警流
其他业务节点 → 本地告警检测 → 写入 Redis 告警流
                            ↓
                    DGX 告警聚合中心（消费者）
                            ↓
          ┌───────────┬───────────┬───────────┐
          去重聚合    分级判定    升级计时
          ↓           ↓           ↓
      企业微信通知  邮件通知    短信/电话升级
```

- **设计原则**：本地检测、集中聚合、分级通知、超时升级、全程可追溯
- **复用基础设施**：Redis Stream（A2A 协议同一套）、企业微信机器人（已有 webhook）
- **对齐规范**：四级告警级别，对应运维响应时效与升级路径

### 1.2 告警分级与升级规则

| 级别 | 触发场景 | 首次通知渠道 | 升级阈值 | 升级后渠道 | 响应时效 |
| --- | --- | --- | --- | --- | --- |
| `info` | 批次成功、系统巡检 | 企业微信（运维群） | - | - | 工作日处理 |
| `warning` | 评分偏低、参数异常 | 企业微信（运维群） | 2小时未处理 | error | 2小时内响应 |
| `error` | 批次失败、脚本异常 | 企业微信（@对应负责人） | 30分钟未处理 | critical | 30分钟内响应 |
| `critical` | 节点离线、NAS挂载失败、连续3次失败 | 企业微信@所有人 + 邮件 | 15分钟未处理 | 短信电话 | 15分钟内响应 |

### 1.3 完整聚合中心代码

**部署位置**：DGX 节点 2，路径 `/data/ops/alert_aggregator.py`

```
#!/usr/bin/env python3
# ==============================================================
# H3 多节点告警聚合与升级中心 v1.0
# 对接：Redis Stream 告警流 + 企业微信 + 邮件升级
# 功能：去重聚合、分级判定、超时升级、多渠道通知
# ==============================================================
import os
import json
import time
import smtplib
import threading
from email.mime.text import MIMEText
from datetime import datetime, timedelta
import redis

# -------------------------- 配置 --------------------------
CONFIG = {
    "redis_host": "10.0.0.12",
    "redis_port": 6379,
    "alert_stream": "stream:ops:alert",
    "consumer_group": "group-alert-center",
    "consumer_name": "consumer-alert-center-01",

    # 企业微信 webhook
    "wecom_webhook": "https://qyapi.weixin.qq.com/cgi-bin/webhook/send?key=你的机器人key",

    # 邮件配置（升级用）
    "smtp_server": "smtp.0379.world",
    "smtp_port": 465,
    "smtp_user": "alert@0379.world",
    "smtp_pass": "授权码",
    "alert_mail_to": ["ops@0379.world", "tech@0379.world"],

    # 升级规则（秒）
    "upgrade_rule": {
        "warning": {"upgrade_to": "error", "timeout": 7200},    # 2小时
        "error": {"upgrade_to": "critical", "timeout": 1800},   # 30分钟
        "critical": {"upgrade_to": "phone", "timeout": 900}     # 15分钟
    },

    # 去重窗口（秒）：同节点同类型同批次
    "dedup_window": 600,

    # 告警状态存储key
    "state_key": "ops:alert:state"
}

redis_client = redis.Redis(
    host=CONFIG["redis_host"],
    port=CONFIG["redis_port"],
    decode_responses=True
)

# -------------------------- 通知方法 --------------------------
def send_wecom(title: str, content: str, level: str = "error", at_users: list = None):
    """发送企业微信，支持@人"""
    color_map = {
        "info": "info",
        "warning": "comment",
        "error": "warning",
        "critical": "warning"
    }

    at_text = ""
    if at_users:
        at_text = "\n" + " ".join([f"<@{u}>" for u in at_users])

    msg = {
        "msgtype": "markdown",
        "markdown": {
            "content": f"""<font color="{color_map.get(level, 'warning')}">**{title}**</font>

> **时间**：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
> **级别**：{level.upper()}

{content}{at_text}
"""
        }
    }

    try:
        import urllib.request
        data = json.dumps(msg).encode("utf-8")
        req = urllib.request.Request(CONFIG["wecom_webhook"], data=data, headers={"Content-Type": "application/json"})
        with urllib.request.urlopen(req, timeout=10) as resp:
            return json.loads(resp.read()).get("errcode") == 0
    except Exception as e:
        print(f"[通知] 企业微信发送失败：{e}")
        return False

def send_mail(subject: str, content: str):
    """发送邮件告警（升级用）"""
    try:
        msg = MIMEText(content, "plain", "utf-8")
        msg["Subject"] = subject
        msg["From"] = CONFIG["smtp_user"]
        msg["To"] = ",".join(CONFIG["alert_mail_to"])

        server = smtplib.SMTP_SSL(CONFIG["smtp_server"], CONFIG["smtp_port"])
        server.login(CONFIG["smtp_user"], CONFIG["smtp_pass"])
        server.sendmail(CONFIG["smtp_user"], CONFIG["alert_mail_to"], msg.as_string())
        server.quit()
        return True
    except Exception as e:
        print(f"[通知] 邮件发送失败：{e}")
        return False

# -------------------------- 聚合与升级逻辑 --------------------------
class AlertAggregator:
    def __init__(self):
        self.running = False
        self.alert_states = {}  # 活跃告警状态：用于升级计时
        self._ensure_consumer_group()
        self._load_states()

    def _ensure_consumer_group(self):
        try:
            redis_client.xgroup_create(
                CONFIG["alert_stream"],
                CONFIG["consumer_group"],
                id="0",
                mkstream=True
            )
        except redis.exceptions.ResponseError as e:
            if "BUSYGROUP" not in str(e):
                raise

    def _load_states(self):
        """加载持久化的告警状态"""
        raw = redis_client.get(CONFIG["state_key"])
        if raw:
            self.alert_states = json.loads(raw)
        else:
            self.alert_states = {}

    def _save_states(self):
        """持久化告警状态"""
        redis_client.set(CONFIG["state_key"], json.dumps(self.alert_states, ensure_ascii=False))

    def _dedup(self, alert_id: str) -> bool:
        """去重检查：True 表示重复"""
        dedup_key = f"ops:alert:dedup:{alert_id}"
        if redis_client.set(dedup_key, "1", ex=CONFIG["dedup_window"], nx=True):
            return False
        return True

    def _upgrade_check(self):
        """定时检查超时告警，执行升级"""
        now = time.time()
        for alert_id, state in list(self.alert_states.items()):
            if state["status"] != "pending":
                continue

            elapsed = now - state["first_time"]
            rule = CONFIG["upgrade_rule"].get(state["level"])
            if not rule:
                continue

            if elapsed >= rule["timeout"] and not state.get("upgraded"):
                # 执行升级
                new_level = rule["upgrade_to"]
                state["level"] = new_level
                state["upgraded"] = True

                title = f"⚠️ 告警升级：{state['title']}"
                content = f"""
> 原始告警ID：{alert_id}
> 原始级别：{state['original_level']}
> 已等待：{int(elapsed/60)} 分钟
> 升级后级别：{new_level.upper()}

请立即处理！
"""
                send_wecom(title, content, new_level, at_users=["all"])

                if new_level == "critical":
                    send_mail(f"[CRITICAL] {state['title']}", content)

                print(f"[升级] 告警 {alert_id} 升级为 {new_level}")

        self._save_states()

    def process_alert(self, alert_data: dict):
        """处理单条告警"""
        alert_id = alert_data.get("alert_id", f"{alert_data.get('node')}-{alert_data.get('batch_id', 'sys')}-{int(time.time())}")

        # 去重
        if self._dedup(alert_id):
            print(f"[去重] 重复告警：{alert_id}")
            return

        level = alert_data.get("level", "error")
        title = alert_data.get("title", "生产告警")
        content = alert_data.get("content", "")
        node = alert_data.get("node", "unknown")

        # 记录状态用于升级
        self.alert_states[alert_id] = {
            "title": title,
            "level": level,
            "original_level": level,
            "node": node,
            "first_time": time.time(),
            "status": "pending",
            "upgraded": False
        }
        self._save_states()

        # 首次通知
        at_users = ["all"] if level in ["error", "critical"] else None
        send_wecom(title, content, level, at_users)

        if level == "critical":
            send_mail(f"[CRITICAL] {title}", content)

        print(f"[告警] {level} | {title} | 节点：{node}")

    def _upgrade_loop(self):
        """升级检查后台线程"""
        while self.running:
            try:
                self._upgrade_check()
            except Exception as e:
                print(f"[升级检查] 异常：{e}")
            time.sleep(60)

    def start(self):
        """启动聚合中心"""
        self.running = True
        threading.Thread(target=self._upgrade_loop, daemon=True).start()
        print("[告警中心] 聚合服务启动")

        while self.running:
            try:
                messages = redis_client.xreadgroup(
                    groupname=CONFIG["consumer_group"],
                    consumername=CONFIG["consumer_name"],
                    streams={CONFIG["alert_stream"]: ">"},
                    count=10,
                    block=2000
                )

                if messages:
                    for stream, msg_list in messages:
                        for msg_id, msg_data in msg_list:
                            try:
                                alert_data = json.loads(msg_data.get("payload", "{}"))
                                self.process_alert(alert_data)
                                redis_client.xack(CONFIG["alert_stream"], CONFIG["consumer_group"], msg_id)
                            except Exception as e:
                                print(f"[处理告警] 异常：{e}")

            except Exception as e:
                print(f"[拉取告警] 异常：{e}")
                time.sleep(1)

# -------------------------- 生产节点上报方法 --------------------------
# 嵌入到各节点本地 alert_notify.py，替换本地发送，改为上报聚合中心
def report_to_aggregator(title: str, content: str, level: str = "error", batch_id: str = None):
    """生产节点上报告警到聚合中心"""
    alert_data = {
        "title": title,
        "content": content,
        "level": level,
        "node": CONFIG["node_name"],
        "batch_id": batch_id,
        "time": datetime.now().isoformat()
    }
    redis_client.xadd(CONFIG["alert_stream"], {"payload": json.dumps(alert_data, ensure_ascii=False)})
```
