# 密钥自动轮换脚本设计

**100% 复用现有基础设施与协议体系**，密钥轮换无缝对接 Nginx 鉴权配置，告警聚合复用 Redis Stream 消息通道，符合「五高架构」高安全、高可用、可审计规范。

---

## 一、API 密钥自动轮换脚本

### 1.1 定位与设计规范

- **定位**：生产级密钥生命周期管理工具，对接 `/h3-dashboard/` 网关鉴权体系，实现密钥自动生成、平滑轮换、过渡期保留、旧密钥回收、全链路审计
- **遵循规范**：
  - 轮换周期：默认 90 天，管理员/只读密钥独立轮换
  - 过渡期：旧密钥保留 7 天，平滑切换无业务中断
  - 双备份：配置文件本地 + NAS RAID1 双副本归档
  - 可审计：每次轮换生成审计日志，写入 NAS 永久留存
  - 零中断：Nginx 平滑重载，业务请求无感知

### 1.2 完整脚本

**部署位置**：DGX 节点 2（网关所在节点），路径 `/data/ops/rotate_api_key.py`

```
#!/usr/bin/env python3
# ==============================================================
# H3 数字人面板 API Key 自动轮换脚本 v1.0
# 对接：Nginx map 鉴权体系 + NAS RAID1 双备份
# 规范：90天轮换周期 + 7天过渡期 + 全程审计可追溯
# ==============================================================
import os
import sys
import json
import time
import uuid
import argparse
import subprocess
from datetime import datetime, timedelta

# -------------------------- 配置 --------------------------
CONFIG = {
    # Nginx 配置文件路径（包含密钥 map 的配置文件）
    "nginx_config_path": "/etc/nginx/conf.d/h3-dashboard-key.conf",
    # Nginx 重载命令
    "nginx_reload_cmd": "nginx -s reload",
    # 本地备份目录
    "backup_dir": "/data/ops/key-backup",
    # NAS 归档目录（RAID1 高可用区）
    "nas_archive_dir": "/mnt/nas/raid1-core/h3-security/key-archive/",
    # 密钥前缀规范
    "key_prefix": {
        "admin": "YYC3-H3-Admin",
        "readonly": "YYC3-H3-Readonly"
    },
    # 轮换周期（天）
    "rotate_days": 90,
    # 过渡期（天），旧密钥保留时间
    "transition_days": 7,
    # 密钥随机串长度
    "key_random_len": 12
}

# -------------------------- 工具函数 --------------------------
def generate_key(key_type: str) -> str:
    """生成符合规范的密钥：前缀-日期-随机串"""
    date_str = datetime.now().strftime("%Y%m%d")
    random_str = uuid.uuid4().hex[:CONFIG["key_random_len"]]
    return f"{CONFIG['key_prefix'][key_type]}-{date_str}-{random_str}"

def load_current_keys() -> dict:
    """解析当前 Nginx 配置中的密钥列表"""
    keys = {"admin": [], "readonly": []}
    if not os.path.exists(CONFIG["nginx_config_path"]):
        return keys

    with open(CONFIG["nginx_config_path"], "r", encoding="utf-8") as f:
        content = f.read()

    # 解析 map 块中的密钥
    in_admin = False
    in_readonly = False
    for line in content.split("\n"):
        line = line.strip()
        if 'map $http_x_api_key $h3_access_granted' in line:
            in_admin = True
            continue
        if in_admin and '}' in line:
            in_admin = False
            continue
        if in_admin and line.startswith('"'):
            key = line.split('"')[1]
            if key and key != "default":
                keys["admin"].append(key)

    # 解析 URL 参数密钥 map（同步轮换）
    in_param = False
    for line in content.split("\n"):
        line = line.strip()
        if 'map $arg_api_key $h3_access_granted_param' in line:
            in_param = True
            continue
        if in_param and '}' in line:
            in_param = False
            continue
        if in_param and line.startswith('"'):
            key = line.split('"')[1]
            if key and key != "default":
                if CONFIG["key_prefix"]["admin"] in key:
                    keys["admin"].append(key)
                elif CONFIG["key_prefix"]["readonly"] in key:
                    keys["readonly"].append(key)

    # 去重
    keys["admin"] = list(dict.fromkeys(keys["admin"]))
    keys["readonly"] = list(dict.fromkeys(keys["readonly"]))
    return keys

def generate_nginx_config(keys: dict) -> str:
    """生成标准 Nginx 密钥 map 配置"""
    config = f"""# ==============================================================
# H3 数字人面板 API Key 鉴权配置
# 自动生成时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
# 轮换周期：{CONFIG['rotate_days']}天 | 过渡期：{CONFIG['transition_days']}天
# ==============================================================

map $http_x_api_key $h3_access_granted {{
    default 0;
"""
    for key in keys["admin"]:
        config += f'    "{key}" 1;\n'
    config += f"}}

map $arg_api_key $h3_access_granted_param {{
    default 0;
"""
    for key in keys["admin"]:
        config += f'    "{key}" 1;\n'
    for key in keys["readonly"]:
        config += f'    "{key}" 1;\n'
    config += "}\n"
    return config

def backup_config():
    """备份当前配置，本地 + NAS 双备份"""
    os.makedirs(CONFIG["backup_dir"], exist_ok=True)
    os.makedirs(CONFIG["nas_archive_dir"], exist_ok=True)

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_file = os.path.join(CONFIG["backup_dir"], f"h3-key-conf.{timestamp}.bak")

    if os.path.exists(CONFIG["nginx_config_path"]):
        with open(CONFIG["nginx_config_path"], "r", encoding="utf-8") as f:
            content = f.read()
        with open(backup_file, "w", encoding="utf-8") as f:
            f.write(content)

        # NAS 归档
        nas_file = os.path.join(CONFIG["nas_archive_dir"], f"h3-key-conf.{timestamp}.bak")
        with open(nas_file, "w", encoding="utf-8") as f:
            f.write(content)

    return backup_file

def clean_expired_keys(keys: dict) -> dict:
    """清理过期密钥：保留过渡期内的旧密钥，删除超期的"""
    cutoff_date = datetime.now() - timedelta(days=CONFIG["transition_days"])
    cleaned = {"admin": [], "readonly": []}

    for key_type in ["admin", "readonly"]:
        for key in keys[key_type]:
            try:
                # 解析密钥中的日期
                date_part = key.split("-")[3]
                key_date = datetime.strptime(date_part, "%Y%m%d")
                if key_date >= cutoff_date:
                    cleaned[key_type].append(key)
            except:
                # 格式异常的密钥保留，人工处理
                cleaned[key_type].append(key)

    return cleaned

def write_audit_log(key_type: str, old_keys: list, new_key: str):
    """写入审计日志，NAS 永久留存"""
    audit_entry = {
        "time": datetime.now().isoformat(),
        "action": "rotate",
        "key_type": key_type,
        "new_key_tail": new_key[-8:],
        "retired_keys": [k[-8:] for k in old_keys if CONFIG["key_prefix"][key_type] in k],
        "node": "dgx-gw-02"
    }

    audit_file = os.path.join(CONFIG["nas_archive_dir"], "key-rotate-audit.log")
    with open(audit_file, "a", encoding="utf-8") as f:
        f.write(json.dumps(audit_entry, ensure_ascii=False) + "\n")

# -------------------------- 主逻辑 --------------------------
def rotate_key(key_type: str, force: bool = False, dry_run: bool = False) -> dict:
    """
    执行密钥轮换
    :param key_type: admin / readonly
    :param force: 强制轮换，忽略周期
    :param dry_run: 预览模式，不实际写入
    """
    current_keys = load_current_keys()
    print(f"当前 {key_type} 密钥数量：{len(current_keys[key_type])}")

    # 检查是否需要轮换
    latest_date = None
    for key in current_keys[key_type]:
        try:
            date_part = key.split("-")[3]
            key_date = datetime.strptime(date_part, "%Y%m%d")
            if not latest_date or key_date > latest_date:
                latest_date = key_date
        except:
            continue

    if not force and latest_date:
        days_since = (datetime.now() - latest_date).days
        if days_since < CONFIG["rotate_days"]:
            print(f"最新密钥生成于 {latest_date.strftime('%Y-%m-%d')}，距今天数 {days_since}，未到轮换周期 {CONFIG['rotate_days']} 天，跳过")
            return {"status": "skipped", "days_since": days_since}

    # 备份当前配置
    backup_file = backup_config()
    print(f"配置已备份：{backup_file}")

    # 清理过期旧密钥
    cleaned_keys = clean_expired_keys(current_keys)
    print(f"清理过期密钥后，剩余 {len(cleaned_keys[key_type])} 个")

    # 生成新密钥
    new_key = generate_key(key_type)
    cleaned_keys[key_type].insert(0, new_key)
    print(f"生成新密钥：{new_key[:8]}...{new_key[-4:]}")

    # 生成配置
    new_config = generate_nginx_config(cleaned_keys)

    if dry_run:
        print("\n[预览模式] 生成的配置如下：")
        print(new_config)
        return {"status": "dry_run", "new_key": new_key}

    # 写入配置
    with open(CONFIG["nginx_config_path"], "w", encoding="utf-8") as f:
        f.write(new_config)

    # 语法校验
    check_result = subprocess.run("nginx -t", shell=True, capture_output=True, text=True)
    if check_result.returncode != 0:
        print(f"❌ Nginx 语法校验失败：{check_result.stderr}")
        # 回滚配置
        subprocess.run(f"cp {backup_file} {CONFIG['nginx_config_path']}", shell=True)
        return {"status": "failed", "error": check_result.stderr}

    # 平滑重载
    reload_result = subprocess.run(CONFIG["nginx_reload_cmd"], shell=True, capture_output=True, text=True)
    if reload_result.returncode != 0:
        print(f"❌ Nginx 重载失败：{reload_result.stderr}")
        return {"status": "failed", "error": reload_result.stderr}

    # 写入审计
    write_audit_log(key_type, current_keys[key_type], new_key)

    print(f"✅ {key_type} 密钥轮换完成，Nginx 已平滑重载")
    return {
        "status": "success",
        "new_key": new_key,
        "backup_file": backup_file
    }

# -------------------------- 命令行入口 --------------------------
if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="H3 面板 API Key 自动轮换工具")
    parser.add_argument("--type", default="readonly", choices=["admin", "readonly", "all"], help="轮换密钥类型")
    parser.add_argument("--force", action="store_true", help="强制轮换，忽略周期")
    parser.add_argument("--dry-run", action="store_true", help="预览模式，不实际修改")
    parser.add_argument("--list", action="store_true", help="列出当前密钥")

    args = parser.parse_args()

    if args.list:
        keys = load_current_keys()
        print("当前密钥列表（仅显示首尾）：")
        for t in ["admin", "readonly"]:
            print(f"\n  {t}:")
            for k in keys[t]:
                print(f"    {k[:12]}...{k[-6:]}")
        sys.exit(0)

    if args.type == "all":
        rotate_key("admin", args.force, args.dry_run)
        print()
        rotate_key("readonly", args.force, args.dry_run)
    else:
        rotate_key(args.type, args.force, args.dry_run)
```
