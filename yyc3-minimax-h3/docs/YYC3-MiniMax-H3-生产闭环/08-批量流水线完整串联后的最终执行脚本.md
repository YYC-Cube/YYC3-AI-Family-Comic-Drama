# 批量流水线完整串联最终执行脚本

**100% 对齐现有网关规范、流水线结构与单一事实源标准**，插入即可形成「API网关→调度→生产→评分→面板→同步→公网访问」完整闭环。

## 定位

`scripts/pipeline-tools/pipeline_full_run.py`，全链路唯一入口脚本，串联**视频生成 → 口型评分 → Manifest 整合 → 面板更新 → NAS 同步触发**全流程，统一参数入口、错误处理、状态回写，严格遵循 `manifest.json` 单一事实源规范。

## 完整代码

```python
#!/usr/bin/env python
# ==============================================================
# H3 数字人批量流水线 最终执行脚本 v2.1
# 全链路：生成 → 评分 → 整合 → 面板 → 同步
# 对齐规范：单一事实源 manifest.json + YYC³ 五高架构
# 对接上游：A2A 生产节点适配器
# ==============================================================
import os
import re
import sys
import json
import argparse
import subprocess
from datetime import datetime

# 路径配置（基于仓库根目录定位）
SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
ROOT_DIR = os.path.abspath(os.path.join(SCRIPT_DIR, "../.."))
OUTPUT_ROOT = os.path.join(ROOT_DIR, "output_batch")

# 子脚本路径
GEN_SCRIPT = os.path.join(ROOT_DIR, "scripts/batch_ref2va_nf4.py")
SCORE_SCRIPT = os.path.join(ROOT_DIR, "scripts/score_lipsync.py")
DASHBOARD_SCRIPT = os.path.join(SCRIPT_DIR, "update_dashboard_batch.py")
# NAS 同步脚本（本地触发，可选）
SYNC_SCRIPT = "/usr/local/bin/sync_h3_core_manual.sh"


def run_cmd(cmd: list, cwd: str = ROOT_DIR) -> tuple:
    """执行命令（argv 数组注入，零 shell 拼接，P0-S2），返回 (成功标志, stdout, stderr)"""
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            capture_output=True,
            text=True,
            timeout=10800  # 单批次上限 3h（生产实测生成约 7000s/条，1800s 必超时，P0-C1）
        )
        return result.returncode == 0, result.stdout, result.stderr
    except subprocess.TimeoutExpired:
        return False, "", "任务执行超时"
    except Exception as e:
        return False, "", str(e)


def load_manifest(batch_dir: str) -> dict:
    """读取 manifest"""
    manifest_path = os.path.join(batch_dir, "manifest.json")
    if not os.path.exists(manifest_path):
        return {}
    with open(manifest_path, "r", encoding="utf-8") as f:
        return json.load(f)


def save_manifest(batch_dir: str, manifest: dict):
    """写入 manifest，单一事实源"""
    manifest_path = os.path.join(batch_dir, "manifest.json")
    manifest["update_time"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(manifest_path, "w", encoding="utf-8") as f:
        json.dump(manifest, f, ensure_ascii=False, indent=2)


def main():
    parser = argparse.ArgumentParser(description="H3 全链路批量生产流水线")
    parser.add_argument("--batch_id", required=True, help="批次唯一ID")
    parser.add_argument("--mode", default="ref2va", choices=["ref2va", "fl2va"], help="生产模式")
    parser.add_argument("--input_text", required=True, help="口播文案")
    parser.add_argument("--ref_image", default="ref_images/default.png", help="参考图路径")
    parser.add_argument("--num_frames", type=int, default=124, help="生成帧数（需满足 %17==5）")
    parser.add_argument("--seed", type=int, default=-1, help="随机种子")
    parser.add_argument("--auto_sync", action="store_true", help="完成后自动触发 NAS 同步")
    parser.add_argument("--output_dir", default=OUTPUT_ROOT, help="输出根目录")

    args = parser.parse_args()
    # 批次 ID 白名单：防路径穿越/注入（P0-S2 配套）
    if not re.fullmatch(r"[0-9A-Za-z_\-]+", args.batch_id):
        parser.error(f"非法批次 ID: {args.batch_id}（仅允许字母数字下划线连字符）")
    batch_dir = os.path.join(args.output_dir, args.batch_id)
    os.makedirs(batch_dir, exist_ok=True)

    # 初始化 manifest
    manifest = {
        "batch_id": args.batch_id,
        "mode": args.mode,
        "input_text": args.input_text,
        "ref_image": args.ref_image,
        "num_frames": args.num_frames,
        "seed": args.seed,
        "status": "running",
        "start_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "start_timestamp": int(datetime.now().timestamp())
    }
    save_manifest(batch_dir, manifest)

    print(f"[流水线启动] 批次：{args.batch_id}，模式：{args.mode}")
    print(f"  输出目录：{batch_dir}")

    # ========== Step1：视频生成 ==========
    print("\n[Step1/5] 执行视频生成...")
    gen_cmd = [
        sys.executable, GEN_SCRIPT,
        "--output_dir", batch_dir,
        "--input_text", args.input_text,
        "--ref_image", args.ref_image,
        "--num_frames", str(args.num_frames),
        "--seed", str(args.seed),
    ]
    success, stdout, stderr = run_cmd(gen_cmd)

    if not success:
        manifest["status"] = "failed"
        manifest["error"] = stderr[-500:]
        save_manifest(batch_dir, manifest)
        # 更新面板失败状态
        run_cmd([sys.executable, DASHBOARD_SCRIPT, "--batch_id", args.batch_id,
                 "--batch_dir", batch_dir, "--status", "failed", "--mode", args.mode])
        print(f"❌ 视频生成失败：{stderr[-200:]}")
        sys.exit(1)

    # 读取生成后的基础 manifest
    manifest = load_manifest(batch_dir)
    manifest["status"] = "generating_done"
    save_manifest(batch_dir, manifest)
    print("✅ 视频生成完成")

    # ========== Step2：口型评分 ==========
    print("\n[Step2/5] 执行 SyncNet 口型评分...")
    score_cmd = [
        sys.executable, SCORE_SCRIPT,
        "--video", os.path.join(batch_dir, "output.mp4"),
        "--audio", os.path.join(batch_dir, "audio.wav"),
        "--output", batch_dir,
    ]
    success, stdout, stderr = run_cmd(score_cmd)

    if success:
        # 解析评分结果，写入 manifest
        try:
            score_result = json.loads(stdout)
            manifest["sync_score"] = score_result.get("sync_score", 0)
            manifest["score_detail"] = score_result
        except:
            manifest["sync_score"] = 0.0
            manifest["score_note"] = "评分解析失败，已降级"
    else:
        manifest["sync_score"] = 0.0
        manifest["score_error"] = stderr[-200:]

    manifest["status"] = "scoring_done"
    save_manifest(batch_dir, manifest)
    print(f"✅ 口型评分完成，同步分：{manifest.get('sync_score', 0)}")

    # ========== Step3：计算最终指标 ==========
    print("\n[Step3/5] 整合最终指标...")
    start_ts = manifest.get("start_timestamp", 0)
    total_seconds = round(datetime.now().timestamp() - start_ts, 2)
    manifest["total_seconds"] = total_seconds
    manifest["avg_frame_seconds"] = round(total_seconds / args.num_frames, 4) if args.num_frames else 0
    manifest["status"] = "completed"
    manifest["finish_time"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    save_manifest(batch_dir, manifest)
    print(f"✅ 指标整合完成，总耗时：{total_seconds}s")

    # ========== Step4：更新面板数据 ==========
    print("\n[Step4/5] 更新可视化面板...")
    success, stdout, stderr = run_cmd(
        [sys.executable, DASHBOARD_SCRIPT, "--batch_id", args.batch_id,
         "--batch_dir", batch_dir, "--status", "completed", "--mode", args.mode]
    )
    if success:
        print("✅ 面板数据更新完成")
    else:
        print(f"⚠️  面板更新异常：{stderr[-100:]}")

    # ========== Step5：触发 NAS 同步（可选） ==========
    if args.auto_sync:
        print("\n[Step5/5] 触发 NAS 核心资产同步...")
        success, stdout, stderr = run_cmd(["bash", SYNC_SCRIPT])
        if success:
            print("✅ NAS 同步已触发")
        else:
            print(f"⚠️  同步触发异常：{stderr[-100:]}")
    else:
        print("\n[Step5/5] 跳过 NAS 同步（待定时任务执行）")

    # ========== 最终输出 ==========
    print("\n" + "="*50)
    print("🎉 全链路流水线执行完成")
    print(f"  批次：{args.batch_id}")
    print(f"  状态：{manifest['status']}")
    print(f"  总耗时：{manifest['total_seconds']}s")
    print(f"  同步分：{manifest.get('sync_score', 0)}")
    print(f"  视频：{os.path.join(batch_dir, 'output.mp4')}")
    print("="*50)


if __name__ == "__main__":
    main()
```

## 对接 A2A 生产节点适配器

替换 `h3_production_adapter.py` 中的 `run_h3_batch` 函数，直接调用全链路脚本，实现一次调用跑完所有环节：

```python
def run_h3_batch(task_params):
    """调用全链路流水线脚本（安全版：argv 数组注入，环境配置复用 02 篇 .env）"""
    batch_name = task_params.get("batch_name", f"batch_{int(time.time())}")
    if not re.fullmatch(r"[0-9A-Za-z_\-]+", batch_name):
        return {"success": False, "error": f"非法批次名: {batch_name}"}

    cmd = [
        PYTHON_BIN, os.path.join(H3_ROOT, "scripts/pipeline-tools/pipeline_full_run.py"),
        "--batch_id", batch_name,
        "--mode", task_params.get("mode", "ref2va"),
        "--input_text", str(task_params.get("input_text", "")),
        "--ref_image", str(task_params.get("ref_image", "ref_images/default.png")),
        "--num_frames", str(task_params.get("num_frames", 124)),
        "--seed", str(task_params.get("seed", -1)),
        "--auto_sync",
    ]

    try:
        result = subprocess.run(
            cmd, cwd=H3_ROOT, capture_output=True, text=True, timeout=TASK_TIMEOUT
        )
        if result.returncode != 0:
            return {"success": False, "error": result.stderr[-500:]}

        # 读取最终 manifest
        batch_dir = os.path.join(H3_ROOT, OUTPUT_DIR, batch_name)
        manifest_path = os.path.join(batch_dir, "manifest.json")
        manifest = {}
        if os.path.exists(manifest_path):
            with open(manifest_path, "r", encoding="utf-8") as f:
                manifest = json.load(f)

        return {
            "success": True,
            "batch_name": batch_name,
            "output_path": batch_dir,
            "video_file": os.path.join(batch_dir, "output.mp4"),
            "manifest": manifest
        }

    except Exception as e:
        return {"success": False, "error": str(e)}
```

## 使用方式

### 1. 命令行手动运行

```bash
# 进入仓库根目录
cd /path/to/YYC3-MiniMax-H3

# 执行全链路生产 + 自动同步NAS
python scripts/pipeline-tools/pipeline_full_run.py \
  --batch_id test_20260924_001 \
  --mode ref2va \
  --input_text "欢迎来到言启象限，语枢未来。" \
  --ref_image ref_images/avatar01.png \
  --num_frames 124 \
  --seed 42 \
  --auto_sync
```

### 2. A2A 调度调用

通过 DGX 端编排引擎提交任务，Mac 节点自动拉取并执行全链路流水线，完成后自动回传结果、更新面板、同步 NAS。
